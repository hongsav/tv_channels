    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // เริ่มต้นเครื่องเล่นวิดีโอ Video.js
            const player = videojs('tv-player', {
                fluid: true,
                playbackRates: [0.5, 1, 1.5, 2]
            });

            const channelItems = document.querySelectorAll('.channel-item');
            const currentChannelName = document.getElementById('current-channel-name');

            // เพิ่ม Event Listener ให้กับรายการช่องทั้งหมด
            channelItems.forEach(item => {
                item.addEventListener('click', function() {
                    // เอา class active ออกจากช่องเดิม
                    document.querySelector('.channel-item.active').classList.remove('active');
                    // ใส่ class active ให้ช่องที่ถูกกด
                    this.classList.add('active');

                    // ดึงค่าลิงก์สตรีมและชื่อช่อง
                    const streamUrl = this.getAttribute('data-src');
                    const channelName = this.getAttribute('data-name');

                    // เปลี่ยนซอร์สวิดีโอและเล่นใหม่
                    player.src({
                        src: streamUrl,
                        type: 'application/x-mpegURL'
                    });
                    player.play();

                    // อัปเดตชื่อช่องบนหน้าจอ
                    currentChannelName.textContent = channelName;
                });
            });
        });
    </script>
